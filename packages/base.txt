# Core system: present on every autarchy machine.
#
# Format for every file in packages/: one package per line, with a trailing
# comment saying why it is here. Parsed only by scripts/pkglist.

base            # Arch base meta-package (systemd, pacman, coreutils, glibc, ...)
linux           # primary kernel
linux-lts       # fallback kernel: a second boot entry if a mainline update breaks boot
man-db          # `man`: local documentation is the first stop when something is unclear
man-pages       # Linux and POSIX man pages
less            # pager for man, git, and systemctl output
neovim          # system editor (configured in Phase 7)
pacman-contrib  # paccache timer keeps the package cache from growing forever
reflector       # keeps /etc/pacman.d/mirrorlist fresh; ships reflector.timer (Phase 9)
