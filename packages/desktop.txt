# The graphical session: compositor, session start, and the components a minimal
# Hyprland desktop needs (Phase 4), plus interaction (Phase 5): launcher, clipboard,
# screenshots/recording, status bar, power menu. Standard, independently-replaceable
# tools, not a unified shell (D-0025).

hyprland                     # the Wayland compositor
uwsm                          # starts Hyprland in its own systemd scopes/slices
sddm                          # graphical greeter; autologin into the uwsm-managed session
ghostty                       # terminal, behind the $terminal role (xdg-terminal-exec)
mako                          # notification daemon
hypridle                      # idle detection, dispatches to hyprlock
hyprlock                      # lock screen
hyprpaper                     # wallpaper, IPC-controlled
hyprpolkitagent               # polkit authentication agent, native to Hyprland
pipewire                      # audio/video routing
pipewire-pulse                # PulseAudio-compatible server, for apps that expect one
pipewire-alsa                 # ALSA-compatible plugin, for apps that use ALSA directly
wireplumber                   # session/policy manager for PipeWire
xdg-desktop-portal-hyprland   # portal backend: screenshare, screenshot
xdg-desktop-portal-gtk        # portal backend: file chooser fallback
matugen                       # single palette -> per-tool templates (D-0025)
vulkan-intel                  # Vulkan driver for the Intel iGPU
xdg-terminal-exec             # resolves the $terminal role; AUR (yay), see packages/external.md
fuzzel                         # launcher; also the --dmenu engine behind the power/system menu
waybar                         # status bar, matugen-templated
networkmanager-dmenu           # network picker through fuzzel (bar Wi-Fi icon, SUPER+CTRL+N): list, join, password, hidden (Phase 17, D-0071)
nm-connection-editor           # full network settings, the icon's right-click: VPN, static IP, 802.1X (a GTK3 window)
cliphist                       # clipboard history
wl-clipboard                   # wl-copy/wl-paste, the primitive cliphist and capture scripts use
grim                            # screenshot capture
slurp                           # region/window selection for screenshots and recording
hyprpicker                      # color picker; also the freeze overlay for screenshots
satty                           # screenshot annotation
gpu-screen-recorder              # screen recording; no Vulkan dependency (checked, D-0032's Phase 5
                                  # revisit) -- uses VAAPI via mesa + intel-media-driver on Intel
intel-media-driver                # VAAPI for gpu-screen-recorder on this Intel iGPU (Broadwell+)
linux-firmware-intel               # optional dep of gpu-screen-recorder (D-0036); installed since
                                    # Phase 5 but never declared -- found by Phase 8's pkg-audit
zbar                              # QR code decoding for capture
libnotify                          # notify-send, for qr-capture's success/failure feedback

# Default browser (Phase 11): closes a real gap -- Phase 5's webapp-install/
# webapp-launch assumed a browser would exist by the time anyone used them, but
# none ever did. Chromium specifically: webapp-launch's app-mode (--app=<url>)
# only works with a Chromium-family browser.
chromium                           # default browser; also drives webapp-launch's app-mode windows
xdg-utils                          # xdg-settings/xdg-open; was already installed transitively,
                                    # never declared until now (a real pkg-audit gap)
xdg-user-dirs                      # creates ~/Documents, ~/Pictures, ...; which ones is system/xdg/user-dirs.defaults

# Visual system (Phase 6): fonts, icons, cursor, GTK/Qt theming -- see D-0037 onward.
ttf-jetbrains-mono-nerd            # terminal/bar/UI font, matches the fontconfig override
papirus-icon-theme                 # icon theme (Papirus-Dark)
bibata-cursor-theme-bin            # cursor theme (Bibata-Modern-Classic); AUR, prebuilt
                                    # (no native hyprcursor port shipped -- XCursor only,
                                    # confirmed against the real GitHub release assets)

# yay itself, plus its own build prerequisites: none of this belongs in packages/
# tooling.txt, which CI installs with plain `pacman -S` in a container that never
# touches the desktop stack -- an AUR-only name there breaks that command outright
# (found the hard way: CI red on "target not found: yay"). The VM's own bulk install
# uses `yay -S --needed $(scripts/pkglist packages/*.txt)` (D-0025/Phase 4), which
# handles official-repo and AUR packages the same way, so these can live wherever;
# desktop.txt is where the thing they exist for (xdg-terminal-exec) already is.
yay                            # AUR helper; bootstrapped manually (packages/external.md)
go                             # builds yay itself (its only real makedepend, go>=1.24)
fakeroot                       # makepkg needs it to fake file ownership when packaging, for any build
make                           # runs yay's and xdg-terminal-exec's Makefiles
scdoc                          # renders xdg-terminal-exec's man page (a makepkg makedepend of that AUR package)
debugedit                      # makepkg needs it to build debug packages (Arch's current default)
gcc                            # C compiler; yay's build uses cgo (Go's C interop) despite being a Go program
